# DoForMe 鸿蒙技术架构文档

**版本**: v3.0
**更新日期**: 2026-02-14
**适用**: HarmonyOS 5.0+ / API 12+

---

## 📋 架构概述

### 当前问题分析

基于项目现状，存在以下架构问题：

| 问题 | 现状 | 影响 |
|------|------|------|
| **模块职责混乱** | `features/Home`承担所有models/services | 难以维护和扩展 |
| **页面集中堆放** | `entry/pages/`有22个页面 | 模块边界模糊 |
| **Feature模块空置** | chat/task/shop/user只有build目录 | 模块化名存实亡 |
| **分层不清晰** | Service直接包含模拟数据 | 数据层和业务层耦合 |
| **缺少状态管理** | 仅使用@State | 跨组件状态共享困难 |
| **鸿蒙特性未充分利用** | 未用AppStorage/EventHub | 未发挥平台优势 |

## 🔧 核心技术方案

### 1. 状态管理方案

#### 层级状态管理

```typescript
// ===== 应用级状态 (AppStorage) =====
// 用于全局共享状态：用户信息、主题、语言等

// commons/core/src/main/ets/state/AppState.ets
export class AppState {
  static init() {
    // 初始化全局状态
    AppStorage.setOrCreate('currentUser', null);
    AppStorage.setOrCreate('isLoggedIn', false);
    AppStorage.setOrCreate('theme', 'light');
    AppStorage.setOrCreate('unreadCount', 0);
  }
}

// 在Ability中使用
@StorageLink('currentUser') currentUser: UserInfo | null = null;
@StorageLink('isLoggedIn') isLoggedIn: boolean = false;

// ===== 页面级状态 (LocalStorage) =====
// 用于页面内组件共享状态

// features/task/src/main/ets/pages/TaskListPage.ets
let taskPageStorage = new LocalStorage();
taskPageStorage.setOrCreate('selectedTab', 0);
taskPageStorage.setOrCreate('filterStatus', 'all');

@Entry(taskPageStorage)
@Component
struct TaskListPage {
  @LocalStorageLink('selectedTab') selectedTab: number = 0;
  // ...
}

// ===== ViewModel状态 (@Observed/@ObjectLink) =====
// 用于复杂对象的响应式

// features/task/src/main/ets/viewmodels/TaskListVM.ets
@Observed
export class TaskListVM {
  tasks: Task[] = [];
  loading: boolean = false;
  error: string = '';
  currentFilter: TaskFilter = { status: 'all' };

  async loadTasks() {
    this.loading = true;
    try {
      const repo = DI.get<ITaskRepository>('ITaskRepository');
      this.tasks = await repo.getTasks(this.currentFilter);
    } catch (e) {
      this.error = e.message;
    } finally {
      this.loading = false;
    }
  }
}

// 在组件中使用
@Component
struct TaskListPage {
  @ObjectLink vm: TaskListVM;

  build() {
    // 自动响应vm属性变化
  }
}
```

#### 状态管理最佳实践

| 状态类型 | 装饰器 | 适用场景 |
|----------|--------|----------|
| 组件私有状态 | `@State` | 仅组件内部使用 |
| 父→子单向传递 | `@Prop` | 子组件只读使用 |
| 父↔子双向绑定 | `@Link` | 子组件需修改 |
| 跨组件共享(页面内) | `@LocalStorageLink` | 页面内多组件共享 |
| 全局共享 | `@StorageLink` | 登录态、主题等 |
| 复杂对象响应式 | `@Observed`+`@ObjectLink` | ViewModel |
| 跨Ability共享 | `@StorageProp`+EventHub | 多窗口场景 |

```typescript
export class DIContainer {

    if (!DIContainer.instance) {
      DIContainer.instance = new DIContainer();

    this.services.set(key, service as object);
  }

  get<T>(key: string): T {
    const service = this.services.get(key);
    if (!service) {

    return service as T;
```

### 4. 网络层方案

```typescript
// commons/data/src/main/ets/remote/ApiClient.ets
import { http } from '@kit.NetworkKit';

export interface ApiResponse<T> {
  code: number;
  message: string;
  data: T;
}

export class ApiClient {
  private static instance: ApiClient;
  private baseUrl: string = 'https://api.doforme.com';
  private httpRequest: http.HttpRequest;

  private constructor() {
    this.httpRequest = http.createHttp();
  }

  static getInstance(): ApiClient {
    if (!ApiClient.instance) {
      ApiClient.instance = new ApiClient();
    }
    return ApiClient.instance;
  }

  async request<T>(
    method: http.RequestMethod,
    path: string,
    options?: {
      params?: Record<string, string>;
      body?: object;
      headers?: Record<string, string>;
    }
  ): Promise<T> {
    const url = this.buildUrl(path, options?.params);
    const token = AppStorage.get<string>('token');

    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
      ...options?.headers
    };
    if (token) {
      headers['Authorization'] = `Bearer ${token}`;
    }

    try {
      const response = await this.httpRequest.request(url, {
        method,
        header: headers,
        extraData: options?.body ? JSON.stringify(options.body) : undefined
      });

      if (response.responseCode === 401) {
        // Token过期，触发登出
        this.handleUnauthorized();
        throw new Error('登录已过期');
      }

      const result = JSON.parse(response.result as string) as ApiResponse<T>;
      if (result.code !== 0) {
        throw new Error(result.message);
      }
      return result.data;
    } catch (error) {
      Logger.error('ApiClient', `Request failed: ${error}`);
      throw error;
    }
  }

  async get<T>(path: string, params?: Record<string, string>): Promise<T> {
    return this.request<T>(http.RequestMethod.GET, path, { params });
  }

  async post<T>(path: string, body?: object): Promise<T> {
    return this.request<T>(http.RequestMethod.POST, path, { body });
  }

  async put<T>(path: string, body?: object): Promise<T> {
    return this.request<T>(http.RequestMethod.PUT, path, { body });
  }

  async delete<T>(path: string): Promise<T> {
    return this.request<T>(http.RequestMethod.DELETE, path);
  }

  private buildUrl(path: string, params?: Record<string, string>): string {
    let url = `${this.baseUrl}${path}`;
    if (params) {
      const query = Object.entries(params)
        .map(([k, v]) => `${k}=${encodeURIComponent(v)}`)
        .join('&');
      url += `?${query}`;
    }
    return url;
  }

  private handleUnauthorized() {
    AppStorage.set('isLoggedIn', false);
    AppStorage.set('currentUser', null);
    AppStorage.set('token', '');
    Navigator.replace(Routes.LOGIN);
  }
}
```

```typescript
// commons/data/src/main/ets/local/PreferencesManager.ets
import { preferences } from '@kit.ArkData';
import { common } from '@kit.AbilityKit';

export class PreferencesManager {
  private static instance: PreferencesManager;
  private prefs: preferences.Preferences | null = null;
  private readonly STORE_NAME = 'doforme_prefs';

  static getInstance(): PreferencesManager {
    if (!PreferencesManager.instance) {
      PreferencesManager.instance = new PreferencesManager();
    }
    return PreferencesManager.instance;
  }

  async init(context: common.UIAbilityContext): Promise<void> {
    this.prefs = await preferences.getPreferences(context, this.STORE_NAME);
  }

  async getString(key: string, defaultValue: string = ''): Promise<string> {
    return (await this.prefs?.get(key, defaultValue)) as string;
  }

  async setString(key: string, value: string): Promise<void> {
    await this.prefs?.put(key, value);
    await this.prefs?.flush();
  }

  async getObject<T>(key: string): Promise<T | null> {
    const json = await this.getString(key);
    if (!json) return null;
    return JSON.parse(json) as T;
  }

  async setObject<T>(key: string, value: T): Promise<void> {
    await this.setString(key, JSON.stringify(value));
  }

  async remove(key: string): Promise<void> {
    await this.prefs?.delete(key);
    await this.prefs?.flush();
  }

  async clear(): Promise<void> {
    await this.prefs?.clear();
    await this.prefs?.flush();
  }
}

// 预定义的Key
export class PrefsKeys {
  static readonly TOKEN = 'auth_token';
  static readonly USER_INFO = 'user_info';
  static readonly THEME = 'app_theme';
  static readonly LANGUAGE = 'app_language';
  static readonly FIRST_LAUNCH = 'first_launch';
}
```

```typescript
}

export class EventBus {
  // 发送事件
  static emit(eventId: AppEvents, data?: object): void {
    emitter.emit({ eventId }, { data });
  }

  // 订阅事件
  static on(eventId: AppEvents, callback: (data?: object) => void): void {
    emitter.on({ eventId }, (event) => {
      callback(event?.data);
    });
  }

  // 取消订阅
  static off(eventId: AppEvents): void {
    emitter.off(eventId);
  }
}

// 使用示例
// 发送
EventBus.emit(AppEvents.TASK_COMPLETED, { taskId: '123', reward: 50 });

// 接收
aboutToAppear() {
  EventBus.on(AppEvents.TASK_COMPLETED, (data) => {
    // 刷新钱包余额
    this.refreshBalance();
  });
}

aboutToDisappear() {
  EventBus.off(AppEvents.TASK_COMPLETED);
}
```

```typescript
  OUTLINE,
  TEXT
}

export enum ButtonSize {
  SMALL,
  MEDIUM,
  LARGE

@Component
export struct AppButton {
  @Prop text: string = '';
  @Prop type: ButtonType = ButtonType.PRIMARY;
  @Prop size: ButtonSize = ButtonSize.MEDIUM;
  @Prop disabled: boolean = false;
  @Prop loading: boolean = false;
  @Prop icon?: Resource;
  onClick: () => void = () => {};

  build() {
    Button({ type: ButtonType.Capsule }) {
      Row({ space: 8 }) {
        if (this.loading) {
          LoadingProgress()
            .width(20)
            .height(20)
            .color(this.getTextColor())
        } else if (this.icon) {
          Image(this.icon)
            .width(20)
            .height(20)
            .fillColor(this.getTextColor())
        }
        Text(this.text)
          .fontSize(this.getFontSize())
          .fontColor(this.getTextColor())
      }
    }
    .width('100%')
    .height(this.getHeight())
    .backgroundColor(this.getBackgroundColor())
    .borderWidth(this.type === ButtonType.OUTLINE ? 1 : 0)
    .borderColor(Colors.primary)
    .enabled(!this.disabled && !this.loading)
    .opacity(this.disabled ? 0.5 : 1)
    .onClick(() => this.onClick())
  }

  private getBackgroundColor(): ResourceColor {
    switch (this.type) {
      case ButtonType.PRIMARY:
        return Colors.primary;
      case ButtonType.SECONDARY:
        return Colors.secondary;
      case ButtonType.OUTLINE:
      case ButtonType.TEXT:
        return Color.Transparent;
      default:
        return Colors.primary;
    }
  }

  private getTextColor(): ResourceColor {
    switch (this.type) {
      case ButtonType.PRIMARY:
        return Color.White;
      case ButtonType.SECONDARY:
        return Colors.primary;
      case ButtonType.OUTLINE:
      case ButtonType.TEXT:
        return Colors.primary;
      default:
        return Color.White;
    }
  }

  private getHeight(): Length {
    switch (this.size) {
      case ButtonSize.SMALL: return 32;
      case ButtonSize.MEDIUM: return 44;
      case ButtonSize.LARGE: return 52;
      default: return 44;
    }
  }

  private getFontSize(): number {
    switch (this.size) {
      case ButtonSize.SMALL: return 12;
      case ButtonSize.MEDIUM: return 14;
      case ButtonSize.LARGE: return 16;
      default: return 14;
    }
  }
}
```

```typescript
      // 内容区域
      if (this.vm.loading) {

      } else if (this.vm.tasks.length === 0) {
```

```typescript
// features/task/src/main/ets/viewmodels/TaskListVM.ets
import { Task, TaskFilter, ITaskRepository } from 'core';
import { DI } from 'core';

@Observed
export class TaskListVM {
  tasks: Task[] = [];
  loading: boolean = false;
  error: string = '';
  private filter: TaskFilter = { status: 'pending' };
  private taskRepo: ITaskRepository;

  constructor() {
    this.taskRepo = DI.get<ITaskRepository>('ITaskRepository');
  }

  async loadTasks(): Promise<void> {
    this.loading = true;
    this.error = '';
    try {
      this.tasks = await this.taskRepo.getTasks(this.filter);
    } catch (e) {
      this.error = (e as Error).message;
    } finally {
      this.loading = false;
    }
  }

  setFilter(filter: Partial<TaskFilter>): void {
    this.filter = { ...this.filter, ...filter };
    this.loadTasks();
  }

  async acceptTask(taskId: string): Promise<boolean> {
    try {
      await this.taskRepo.acceptTask(taskId);
      await this.loadTasks();
      return true;
    } catch (e) {
      this.error = (e as Error).message;
      return false;
    }
  }
}
```

```typescript
import { abilityAccessCtrl, Permissions } from '@kit.AbilityKit';

export class PermissionUtils {
  static async checkPermission(
    context: common.UIAbilityContext,
    permission: Permissions
  ): Promise<boolean> {
    const atManager = abilityAccessCtrl.createAtManager();
    const grantStatus = await atManager.checkAccessToken(
      context.applicationInfo.accessTokenId,
      permission
    );
    return grantStatus === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED;
  }

  static async requestPermission(
    context: common.UIAbilityContext,
    permission: Permissions
  ): Promise<boolean> {
    const atManager = abilityAccessCtrl.createAtManager();
    const result = await atManager.requestPermissionsFromUser(
      context,
      [permission]
    );
    return result.authResults[0] === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED;
  }
}
```

```typescript
// 使用LazyForEach + IDataSource
class TaskDataSource implements IDataSource {
  private tasks: Task[] = [];
  private listeners: DataChangeListener[] = [];

  totalCount(): number {
    return this.tasks.length;
  }

  getData(index: number): Task {
    return this.tasks[index];
  }

  setData(tasks: Task[]): void {
    this.tasks = tasks;
    this.notifyDataReload();
  }

  registerDataChangeListener(listener: DataChangeListener): void {
    this.listeners.push(listener);
  }

  unregisterDataChangeListener(listener: DataChangeListener): void {
    const index = this.listeners.indexOf(listener);
    if (index >= 0) {
      this.listeners.splice(index, 1);
    }
  }

  private notifyDataReload(): void {
    this.listeners.forEach(listener => listener.onDataReloaded());
  }
}

// 在组件中使用
@Component
struct TaskList {
  private dataSource: TaskDataSource = new TaskDataSource();

      LazyForEach(this.dataSource, (task: Task) => {

          TaskCard({ task })
        }
      }, (task: Task) => task.taskId)
    }
    .cachedCount(5)  // 缓存数量
  }
```
