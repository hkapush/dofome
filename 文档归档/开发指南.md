## 📊 核心数据模型

### Task（任务）
```typescript
interface Task {
  taskId: string;           // 任务ID
  title: string;            // 任务标题
  description: string;      // 任务描述
  reward: number;           // 悬赏金额
  currencyType: string;     // 货币类型: shell/star/guild
  status: TaskStatus;       // 任务状态
  category: string;         // 任务分类
  deadline: string | number;// 截止时间
  executionMode: ExecutionMode; // 顺序/自由执行
  subTasks: SubTask[];      // 子任务列表
  publisherId: string;      // 发布者ID
  executorId?: string;      // 执行者ID
}

enum TaskStatus {
  PENDING = 'pending',      // 待接取
  IN_PROGRESS = 'in_progress', // 进行中
  COMPLETED = 'completed',  // 已完成
  CANCELLED = 'cancelled'   // 已取消
}

enum ExecutionMode {
  SEQUENTIAL = 'sequential', // 顺序执行
  FREE = 'free'              // 自由执行
}
```

### Product（商品）
```typescript
interface Product {
  productId: string;        // 商品ID
  name: string;             // 商品名称
  description: string;      // 商品描述
  price: number;            // 价格
  currencyType: CurrencyType; // 接受的货币类型
  category: ProductCategory;  // 商品分类
  stock: number;            // 库存数量
  sales: number;            // 销量
  rating: number;           // 评分
  storeId: string;          // 所属店铺ID
}
```

### Message（消息）
```typescript
interface Message {
  messageId: string;        // 消息ID
  type: MessageType;        // 消息类型
  title: string;            // 消息标题
  content: string;          // 消息内容
  isRead: boolean;          // 是否已读
  createdAt: string;        // 创建时间
}

enum MessageType {
  SYSTEM = 'system',        // 系统消息
  TASK = 'task',            // 任务消息
  ORDER = 'order',          // 订单消息
  CHAT = 'chat'             // 聊天消息
}
```

### 时间格式化
```typescript
function formatTime(timestamp: number): string {
  const now = Date.now();
  const diff = now - timestamp;

  if (diff < 60000) return '刚刚';
  if (diff < 3600000) return `${Math.floor(diff / 60000)}分钟前`;
  if (diff < 86400000) return `${Math.floor(diff / 3600000)}小时前`;
  return new Date(timestamp).toLocaleDateString();
}
```

### 组件结构
```typescript
@Component
export struct MyComponent {
  // 1. 状态变量
  @State data: string[] = [];
  @State loading: boolean = false;

  // 2. 普通变量
  private service = MyService.getInstance();

  // 3. 生命周期
  aboutToAppear() {
    this.loadData();
  }

  // 4. 业务方法
  async loadData() {
    this.loading = true;
    this.data = await this.service.getData();
    this.loading = false;
  }

  // 5. 构建方法
  build() {
    Column() {
      // UI
    }
  }
}
```
