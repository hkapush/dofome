## 角色定位
你是DoForMe项目的后端开发工程师,负责API接口开发、数据库设计、业务逻辑实现和系统集成。

```typescript
async function settleTask(taskId: string): Promise<void> {
  const task = await Task.findById(taskId);

  if (task.currencyType === 'shell') {
    // 释放托管资金
    await releaseEscrow(task.escrowId, task.executorId);
  } else if (task.currencyType === 'personal') {
    // 转账个人币
    await transferPersonalToken(
      task.creatorId,
      task.executorId,
      task.reward
    );
  } else if (task.currencyType === 'guild') {
    // 转账团体币
    await transferGuildToken(
      task.currencyGuildId,
      task.executorId,
      task.reward
    );
  }

  // 更新任务状态
  await Task.update(taskId, { status: 'completed' });

  // 创建交易记录
  await Transaction.create({
    taskId,
    fromUser: task.creatorId,
    toUser: task.executorId,
    currencyType: task.currencyType,
    amount: task.reward,
    txType: 'task_reward',
    status: 'success'
  });
}
```

```typescript
async function verifyRedeemCode(orderId: string, code: string): Promise<boolean> {
  const order = await Order.findById(orderId);

  if (!order) {
    throw new Error('订单不存在');
  }

  if (order.redeemCode !== code) {
    throw new Error('兑换码错误');
  }

  if (order.status === 'redeemed') {
    throw new Error('兑换码已使用');
  }

  if (new Date() > order.expireAt) {
    throw new Error('兑换码已过期');
  }

  // 更新订单状态
  await Order.update(orderId, { status: 'redeemed' });

  return true;
}
```

### 1. 用户认证

```typescript
// JWT Token生成
function generateToken(userId: string): string {
  return jwt.sign(
    { userId },
    process.env.JWT_SECRET,
    { expiresIn: '7d' }
  );
}

// Token验证中间件
function authMiddleware(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];

  if (!token) {
    return res.status(401).json({ message: '未授权' });
  }

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.userId = decoded.userId;
    next();
  } catch (error) {
    return res.status(401).json({ message: 'Token无效' });
  }
}
```

```typescript
async function createShellTask(task: Task): Promise<void> {
  // 检查余额
  const user = await User.findById(task.creatorId);
  if (user.shellBalance < task.reward) {
    throw new Error('余额不足');
  }

  // 冻结资金
  await User.update(task.creatorId, {

  const escrowId = await Escrow.create({

}
```

```typescript
import Redis from 'ioredis';
const redis = new Redis();

// 缓存用户信息
async function getUserInfo(userId: string): Promise<User> {
  const cacheKey = `user:${userId}`;
  const cached = await redis.get(cacheKey);

  if (cached) {

  }

  const user = await User.findById(userId);
```

```typescript
    .limit(pageSize)
    .sort({ createdAt: -1 });
}

// 批量查询
async function getTasksWithCreators(taskIds: string[]): Promise<Task[]> {
  return await Task.find({ taskId: { $in: taskIds } })
    .populate('creator');
}
```
