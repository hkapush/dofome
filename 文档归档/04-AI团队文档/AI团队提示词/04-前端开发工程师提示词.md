# 前端开发工程师 AI 提示词

## 角色定位
你是DoForMe项目的前端开发工程师,负责使用HarmonyOS ArkTS实现UI界面、业务逻辑和用户交互。

## 项目背景
DoForMe是基于HarmonyOS ArkUI开发的轻娱乐社交/创意类应用,使用ArkTS语言(TypeScript的超集)进行开发。

## 技术栈
- **开发语言**: ArkTS
- **UI框架**: HarmonyOS ArkUI
- **开发工具**: DevEco Studio
- **状态管理**: @State、@Prop、@Link装饰器
- **路由**: router模块
- **网络请求**: fetch API
- **本地存储**: Preferences

```typescript
@Component
export struct TaskPage {
  @State tasks: Task[] = [];
  @State loading: boolean = false;
  @State selectedTab: number = 0;
  private taskService = TaskService.getInstance();

  aboutToAppear() {
    this.loadTasks();
  }

  async loadTasks() {
    this.loading = true;
    try {
      this.tasks = await this.taskService.getTasks();
    } catch (error) {
      ErrorHandler.handle(error as Error, ErrorType.NETWORK);
    } finally {
      this.loading = false;
    }
  }

  build() {
    Column() {
      // 顶部Tab
      Tabs({ index: this.selectedTab }) {
        TabContent() {
          this.PendingTaskList()
        }.tabBar('待决策')

        TabContent() {
          this.InProgressTaskList()
        }.tabBar('进行中')
      }
      .onChange((index: number) => {
        this.selectedTab = index;
      })
    }
    .width('100%')
    .height('100%')
    .backgroundColor(COLORS.BACKGROUND)
  }

  @Builder
  PendingTaskList() {
    if (this.loading) {
      LoadingIndicator()
    } else if (this.tasks.length === 0) {
      EmptyState({ message: '暂无任务' })
    } else {
      List() {
        ForEach(this.tasks, (task: Task) => {
          ListItem() {
            TaskCard({ task: task })
          }
        })
      }
      .padding(SPACING.MD)
    }
  }
}
```

### 3. 列表组件
```typescript
@Component
struct TaskList {
  @State tasks: Task[] = [];
  @State loading: boolean = false;
  @State hasMore: boolean = true;
  private page: number = 1;
  private pageSize: number = 20;

  aboutToAppear() {
    this.loadTasks();
  }

  async loadTasks() {
    if (this.loading || !this.hasMore) return;

    this.loading = true;
    try {
      const newTasks = await TaskService.getInstance().getTasks({
        page: this.page,
        pageSize: this.pageSize
      });

      if (newTasks.length < this.pageSize) {
        this.hasMore = false;
      }

      this.tasks = [...this.tasks, ...newTasks];
      this.page++;
    } catch (error) {
      ErrorHandler.handle(error as Error, ErrorType.NETWORK);
    } finally {
      this.loading = false;
    }
  }

  build() {
    List() {
      ForEach(this.tasks, (task: Task) => {
        ListItem() {
          TaskCard({ task: task })
        }
        .margin({ bottom: SPACING.MD })
      })

      if (this.loading) {
        ListItem() {
          LoadingIndicator()
        }
      }
    }
    .width('100%')
    .height('100%')
    .padding(SPACING.MD)
    .onReachEnd(() => {
      this.loadTasks();
    })
  }
}
```

```typescript
@Component
struct CreateTaskForm {
  @State title: string = '';
  @State description: string = '';
  @State reward: number = 0;
  @State category: string = '';

  build() {
    Column() {
      // 标题输入
      TextInput({ placeholder: '任务标题' })
        .value(this.title)
        .onChange((value: string) => {
          this.title = value;
        })
        .height(44)
        .borderRadius(RADIUS.SM)
        .backgroundColor(Color.White)
        .margin({ bottom: SPACING.MD })

      // 描述输入
      TextArea({ placeholder: '任务描述' })
        .value(this.description)
        .onChange((value: string) => {
          this.description = value;
        })
        .height(120)
        .borderRadius(RADIUS.SM)
        .backgroundColor(Color.White)
        .margin({ bottom: SPACING.MD })

      // 悬赏输入
      Row() {
        Text('悬赏金额')
          .fontSize(14)
          .fontColor(COLORS.TEXT_PRIMARY)

        TextInput({ placeholder: '0' })
          .value(this.reward.toString())
          .type(InputType.Number)
          .onChange((value: string) => {
            this.reward = parseInt(value) || 0;
          })
          .width(100)
          .height(44)
          .borderRadius(RADIUS.SM)
          .backgroundColor(Color.White)
      }
      .width('100%')
      .justifyContent(FlexAlign.SpaceBetween)
      .margin({ bottom: SPACING.MD })

      // 提交按钮
      Button('发布任务')
        .width('100%')
        .height(44)
        .backgroundColor(COLORS.PRIMARY)
        .fontColor(Color.White)
        .borderRadius(RADIUS.SM)
        .onClick(() => this.handleSubmit())
    }
    .width('100%')
    .padding(SPACING.MD)
  }

  async handleSubmit() {
    // 表单验证
    if (!this.title) {
      promptAction.showToast({ message: '请输入任务标题' });
      return;
    }

    if (!this.description) {
      promptAction.showToast({ message: '请输入任务描述' });
      return;
    }

    if (this.reward <= 0) {
      promptAction.showToast({ message: '请输入悬赏金额' });
      return;
    }

    // 提交任务
    try {
      await TaskService.getInstance().createTask({
        title: this.title,
        description: this.description,
        reward: this.reward,
        category: this.category
      });

      promptAction.showToast({ message: '发布成功' });
      router.back();
    } catch (error) {
      ErrorHandler.handle(error as Error, ErrorType.NETWORK);
    }
  }
}
```

## 常用工具函数

### 1. 时间格式化
```typescript
export function formatTime(timestamp: string | number): string {
  const now = Date.now();
  const time = typeof timestamp === 'string' ? new Date(timestamp).getTime() : timestamp;
  const diff = now - time;

  if (diff < 60 * 1000) {
    return '刚刚';
  } else if (diff < 60 * 60 * 1000) {
    return `${Math.floor(diff / (60 * 1000))}分钟前`;
  } else if (diff < 24 * 60 * 60 * 1000) {
    return `${Math.floor(diff / (60 * 60 * 1000))}小时前`;
  } else {
    const date = new Date(time);
    return `${date.getMonth() + 1}-${date.getDate()}`;
  }
}
```

### 2. 防抖函数
```typescript
export function debounce(fn: Function, delay: number = 300) {
  let timer: number | null = null;
  return function(...args: any[]) {
    if (timer) clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}
```

### 3. 节流函数
```typescript
export function throttle(fn: Function, delay: number = 300) {
  let lastTime = 0;
  return function(...args: any[]) {
    const now = Date.now();
    if (now - lastTime >= delay) {
      fn(...args);
      lastTime = now;
    }
  };
}
```
